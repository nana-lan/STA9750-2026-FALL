get_eia_sep <- function(state, abbr){
    library(tidyverse)
    library(stringr)
    library(httr2)
    library(rvest)
    library(fs)
  
    state_formatted <- str_to_lower(state) |> str_replace_all("\\s", "")
    
    dir_name <- fs::path("data", "mp02")
    file_name <- fs::path(dir_name, state_formatted)
    
    fs::dir_create(dir_name, recurse=TRUE)
    
    if(!fs::file_exists(file_name)){
        BASE_URL <- "https://www.eia.gov"
        RESPONSE <- request(BASE_URL) |> 
          req_url_path("electricity", 
                       "state", 
                       state_formatted) |>
          req_headers(
            `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
            ) |>   
          req_retry(
                max_tries = 4, 
          ) |>
          req_perform(file_name)
    
        resp_check_status(RESPONSE)
    }
    
    TABLE <- read_html(file_name) |> 
        html_element("table") |> 
        html_table() |>
        mutate(Item = str_to_lower(Item))
    
    if("U.S. rank" %in% colnames(TABLE)){
        TABLE <- TABLE |> rename(Rank = `U.S. rank`)
    }
    
    CO2_MWh <- TABLE |> 
        filter(Item == "carbon dioxide (lbs/mwh)") |>
        pull(Value) |> 
        str_replace_all(",", "") |>
        as.numeric()
    
    PRIMARY <- TABLE |> 
        filter(Item == "primary energy source") |> 
        pull(Rank)
    
    RATE <- TABLE |>
        filter(Item == "average retail price (cents/kwh)") |>
        pull(Value) |>
        as.numeric()
    
    GENERATION_MWh <- TABLE |>
        filter(Item == "net generation (megawatthours)") |>
        pull(Value) |>
        str_replace_all(",", "") |>
        as.numeric()
    
    data.frame(CO2_MWh               = CO2_MWh, 
               primary_source        = PRIMARY,
               electricity_price_MWh = RATE * 10, # / 100 cents to dollars &
               # * 1000 kWh to MWH 
               generation_MWh        = GENERATION_MWh, 
               state                 = state, 
               abbreviation          = abbr
    )
}

EIA_SEP_REPORT <- map2(state.name, state.abb, get_eia_sep) |> list_rbind()

load_ntd_energy <- function(){
  library(readxl)
  library(httr2)
  library(tidyverse)
  library(fs)
  
  # Create 'data/mp02' directory if not already present
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_ENERGY_FILE <- fs::path(DATA_DIR, "2025_ntd_energy.xlsx")

  if(!fs::file_exists(NTD_ENERGY_FILE)){
    library(httr2)
  
    request("https://www.transit.dot.gov/") |>
      req_url_path("sites",
                   "fta.dot.gov",
                   "files",
                   "2026-09",
                   "2025 Energy Consumption_260806.xlsx") |> 
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_ENERGY_FILE)
  }
  
  if(file.info(NTD_ENERGY_FILE)$size < 1e4){
        cat("I was unable to download the NTD Energy File. Please try again.\n")
        stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_energy.xlsx')` and try again.")
    }

  to_numeric_fill_0 <- function(x){
    x <- if_else(x == "-", NA, x)
    replace_na(as.numeric(x), 0)
  }

  read_excel(NTD_ENERGY_FILE) |> 
    select(-c(`Reporter Type`, 
              `Reporting Module`, 
              `Gallons of Other Fuel`, 
              `Other Fuel Description`)) |>
    mutate(across(-c(`Agency Name`, 
                     `Mode`,
                     `TOS`), 
                  to_numeric_fill_0)) |>
    group_by(`NTD ID`, `Mode`, `Agency Name`) |>
    summarize(across(where(is.numeric), sum), 
              .groups = "keep") |>
    mutate(ENERGY = sum(c_across(c(where(is.numeric))))) |>
    filter(ENERGY > 0) |>
    select(-ENERGY) |>
    ungroup()
}

NTD_ENERGY <- load_ntd_energy()

load_ntd_service <- function(){
  library(httr2)
  library(tidyverse)
  library(fs)
  
  # Create 'data/mp02' directory if not already present
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_SERVICE_FILE <- fs::path(DATA_DIR, "2025_ntd_service.csv")

  if(!fs::file_exists(NTD_SERVICE_FILE)){
    library(httr2)
  
    request("https://data.transportation.gov/resource/4fir-qbim.csv") |>
      req_url_query(`$query`=paste(
        "SELECT  `_5_digit_ntd_id` AS `NTD_ID`", 
        "`max_agency` AS `AGENCY`",
        "`max_reporter_type` AS `reporter_type`",
        "`max_organization_type` AS `organization_type`",
        "`max_city` AS `city`",
        "`max_state` AS `state`",
        "`mode`",
        "`type_of_service`",
        "`max_primary_uza_code` as `uza_code`",
        "`max_primary_uza_name` as `uza_name`",
        "`max_primary_uza_area_sq_miles` as `uza_sq_miles`",
        "`max_primary_uza_population` as `uza_population`",
        "`max_service_area_sq_miles` as `service_area_sq_miles`",
        "`max_service_area_population` as `service_area_population`",
        "`questionable_record` as `flagged_likely_inaccurate`",
        "`sum_unlinked_passenger_trips_upt` as `total_upt`",
        "`sum_passenger_miles` WHERE caseless_one_of(`report_year`, '2025') limit 50000", 
        sep=",")) |>
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_SERVICE_FILE)
  }
  
  if(file.info(NTD_SERVICE_FILE)$size < 1e4){
        cat("I was unable to download the NTD Service File. Please try again.\n")
        stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_service.xlsx')` and try again.")
    }

  read_csv(NTD_SERVICE_FILE) |>
    mutate(flagged_likely_inaccurate = !is.na(flagged_likely_inaccurate))
}

Data Analysis

Calculate Total Emissions

For the carbon emissions analysis, we used a multi-table integration approach.

Show the code 
agency_state <- NTD_SERVICE |>
  select(NTD_ID, state) |>
  distinct()

agency_full <- NTD_ENERGY |>
  mutate(
    `NTD ID` = str_pad(
      as.character(`NTD ID`),
      width = 5,
      side = "left",
      pad = "0"
    )
  ) |>
  left_join(agency_state, by = c("NTD ID" = "NTD_ID")) |>
  left_join(EIA_SEP_REPORT |>
      select(abbreviation, CO2_MWh), by = c("state" = "abbreviation"))

First, we mutate() a column in NTD_ENERGY to make the width of NTD ID correspond with 5 number and left_join() with NTD_SERVICE, and then we joined the state-level electricity carbon intensity data from the U.S. Energy Information Administration (EIA_SEP_REPORT). The final table includes one row for each Agency + Mode pair, along with its state, all fuel consumption data, and the corresponding CO2_MWh.

Show the code 
agency_emissions <- agency_full |>
  mutate(
    emissions_diesel =
      replace_na(`Gallons of Diesel Fuel`, 0) * 10.19,

    emissions_gasoline =
      replace_na(`Gallons of Gasoline`, 0) * 8.49,

    emissions_kerosene =
      replace_na(`Gallons of Kerosene`, 0) * 9.88,

    emissions_lpg =
      replace_na(`Gallons of Liquified Petroleum Gas`, 0) * 5.75,
    
    emissions_bunker =
      replace_na(`Gallons of Bunker Fuel`, 0) * 11.24,
    
    emissions_cng =
      replace_na(`Gallons of Compressed Natural Gas`, 0) * 6.03,
    
    emissions_lng =
      replace_na(`Gallons of Liquified Nat Gas`, 0) * 4.02,

    emissions_elec_prop =
      replace_na(`Kilowatt hours of Electric Propulsion`, 0) /
      1000 * CO2_MWh * 0.453592,

    emissions_elec_batt =
      replace_na(`Kilowatt hours of Electric Battery`, 0) /
      1000 * CO2_MWh * 0.453592,
    
    total_emissions_kg =
      emissions_diesel +
      emissions_gasoline +
      emissions_kerosene +
      emissions_lpg +
      emissions_bunker +
      emissions_cng +
      emissions_lng +
      emissions_elec_prop +
      emissions_elec_batt
  )

To calculate total carbon emissions, we multiplied the consumption of fuels such as diesel and gasoline by their corresponding CO2 emission factors. For example, diesel produces about 10.19 kg of CO2 per gallon. For electricity, we first converted electricity use from kWh to MWh and then multiplied it by the electricity carbon intensity of the state. Finally, emissions from different energy sources were added together to estimate the total_emissions_kg for each Agency + Mode pair.

Relate Total Emissions and Passenger Service

However, total emissions alone cannot fairly reflect the environmental performance of each transit agency. Larger transit agencies usually serve more passengers, so higher total emissions do not necessarily mean lower carbon efficiency. To make the comparison more meaningful, we aggregated emissions to the agency level and used two normalized metrics:

  • Emissions per UPT: measures the average emissions generated for each unlinked passenger trip.
  • Emissions per Passenger Mile: measures the emissions generated for carrying one passenger for one mile.

Show the code 
agency_electrification <- agency_full |>
  group_by(`NTD ID`, `Agency Name`, state) |>
  summarize(
    electric_gge =
      sum(
        replace_na(`Kilowatt hours of Electric Propulsion`, 0) +
        replace_na(`Kilowatt hours of Electric Battery`, 0),
        na.rm = TRUE
      ) * 0.030,

    gasoline_gge =
      sum(replace_na(`Gallons of Gasoline`, 0), na.rm = TRUE),

    diesel_gge =
      sum(replace_na(`Gallons of Diesel Fuel`, 0), na.rm = TRUE) * 1.12,

    biodiesel_gge =
      sum(replace_na(`Gallons of Bio-Diesel`, 0), na.rm = TRUE) * 1.05,

    lpg_gge =
      sum(replace_na(`Gallons of Liquified Petroleum Gas`, 0), na.rm = TRUE) * 0.74,

    hydrogen_gge =
      sum(replace_na(`Kilograms of Hydrogen`, 0), na.rm = TRUE) * 1.00,

    cng_gge =
      sum(replace_na(`Gallons of Compressed Natural Gas`, 0), na.rm = TRUE),

    .groups = "drop"
  ) |>
  mutate(
    total_gge =
      electric_gge +
      gasoline_gge +
      diesel_gge +
      biodiesel_gge +
      lpg_gge +
      hydrogen_gge +
      cng_gge,

    electrification_rate =
      electric_gge / total_gge
  ) |>
  filter(total_gge > 0)

electrification_winners <- agency_electrification |>
  slice_max(
    electric_gge,
    n = 3,
    with_ties = FALSE
  )

We first converted different fuel types into the same energy unit, and then summed the energy use of all transit modes for each agency. Next, we calculated the electrification rate by dividing electric_gge by total_gge, which shows the percentage of total energy use that comes from electricity. Finally, we selected the three agencies with the highest electrification rates.
